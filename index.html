import random # ============ نظام الموارد والفلوس الجديد ============ class EconomySystem: def __init__(self, country): self.country = country # الموارد اللي عندك self.resources = { "oil": 100, # بترول "gold": 50, # دهب "food": 200, # اكل للجيش والشعب "iron": 80 # حديد لصناعة سلاح } self.money = 1000 # فلوسك البدائية self.currency_rate = 1.0 # قوة عملتك vs الدولار self.income_per_turn = 0 def collect_resources(self): """ده اللي بيجيب فلوس كل دور""" # 1. دخل من الموارد - كل مورد ليه سعر prices = {"oil": 5, "gold": 20, "food": 2, "iron": 8} income = 0 for res, amount in self.resources.items(): # كل دور بتطلع 10% من المورد extracted = amount * 0.1 income += extracted * prices[res] self.resources[res] -= extracted * 0.05 # الموارد بتقل شوية # 2. ضرايب من الشعب tax = self.country.power * 2 + self.country.economy * 3 income += tax # 3. مصاريف الجيش army_cost = self.country.army * 0.05 income -= army_cost # 4. دخل من التجارة trade_bonus = 0 for t in self.country.treaties: if t.type == "trade": trade_bonus += 150 income += trade_bonus self.money += income self.income_per_turn = income return income def trade_resource(self, resource, amount, price, other_country_econ): """تبيع مورد لدولة تانية""" if self.resources[resource] < amount: return f"ماعندكش {resource} كفاية!" # قوة عملتك بتأثر على السعر final_price = price * self.currency_rate # لو اقتصاد اللي قدامك قوي هيدفع اكتر if other_country_econ.economy > 80: final_price *= 1.3 self.resources[resource] -= amount self.money += final_price * amount other_country_econ.resources[resource] += amount other_country_econ.money -= final_price * amount return f"بعت {amount} {resource} بـ {final_price*amount:.0f}$" def print_status(self): print(f"\n--- اقتصاد {self.country.name} ---") print(f"💰 الفلوس: {self.money:.0f}$ | الدخل/دور: {self.income_per_turn:.0f}$") print(f"💱 قوة العملة: {self.currency_rate}") print(f"📦 الموارد: {self.resources}") # لو فلوسك قلت العملة بتقع if self.money < 500: self.currency_rate = 0.7 print("⚠️ تحذير: عملتك بتنهار! لازم تبيع موارد او تفرض ضرايب") # ============ مثال تشغيل الاقتصاد ============ class Country: def __init__(self, name, power, economy, army): self.name = name self.power = power self.economy = economy self.army = army self.treaties = [] self.econ_system = EconomySystem(self) # انشاء دولتك egypt = Country("Egypt", 75, 70, 50000) saudi = Country("Saudi", 80, 90, 60000) saudi.econ_system.resources["oil"] = 500 # السعودية بترول كتير # اضافة معاهدة تجارة class Treaty: def __init__(self, type): self.type = type egypt.treaties.append(Treaty("trade")) # --- اللوب بتاع اللعبة كل دور --- for turn in range(1, 6): print(f"\n========== الدور {turn} ==========") income_egypt = egypt.econ_system.collect_resources() income_saudi = saudi.econ_system.collect_resources() egypt.econ_system.print_status() # مثال تبيع بترول if turn == 3: print("\n[قرار ذكي]: بتبيع 20 بترول للسعودية") # هنا هتبيع انت؟ لا السعودية عندها بترول، فهتبيع دهب مثلا result = egypt.econ_system.trade_resource("gold", 10, 20, saudi.econ_system) print(result)
